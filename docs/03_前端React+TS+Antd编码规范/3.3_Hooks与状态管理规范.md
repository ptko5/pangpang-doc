## 2. Hooks使用规范

### 2.1 基础Hooks

#### 【强制】useState

```tsx
function UserForm() {
  const [username, setUsername] = useState<string>('');
  const [isLoading, setIsLoading] = useState<boolean>(false);
  const [errors, setErrors] = useState<Record<string, string>>({});

  const handleSubmit = () => {
    setIsLoading(true);
    // ...
  };

  return (
    <form onSubmit={handleSubmit}>
      <input
        type="text"
        value={username}
        onChange={(e) => setUsername(e.target.value)}
      />
    </form>
  );
}
```

#### 【强制】useEffect

```tsx
function UserList() {
  const [users, setUsers] = useState<User[]>([]);
  const [isLoading, setIsLoading] = useState<boolean>(true);

  useEffect(() => {
    const fetchUsers = async () => {
      setIsLoading(true);
      try {
        const response = await getUserList();
        setUsers(response.data);
      } catch (error) {
        console.error('Failed to fetch users:', error);
      } finally {
        setIsLoading(false);
      }
    };

    fetchUsers();
  }, []);

  return <div>{isLoading ? 'Loading...' : users.map(user => <div key={user.id}>{user.username}</div>)}</div>;
}
```

#### 【强制】useEffect依赖数组

- 依赖数组必须明确列出所有依赖项
- 禁止空依赖数组 `[]` 时使用外部变量

```tsx
function UserDetail({ userId }: { userId: number }) {
  const [user, setUser] = useState<User | null>(null);

  useEffect(() => {
    const fetchUser = async () => {
      const response = await getUserById(userId);
      setUser(response.data);
    };

    fetchUser();
  }, [userId]);

  return <div>{user?.username}</div>;
}
```

#### 【推荐】useContext

```tsx
interface AuthContextType {
  user: User | null;
  login: (username: string, password: string) => Promise<void>;
  logout: () => void;
}

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);

  const login = async (username: string, password: string) => {
    const response = await api.login({ username, password });
    setUser(response.data.user);
    storage.setToken(response.data.token);
  };

  const logout = () => {
    setUser(null);
    storage.removeToken();
  };

  return (
    <AuthContext.Provider value={{ user, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
}
```

#### 【推荐】useMemo/useCallback

```tsx
function UserList({ users, filter }: { users: User[]; filter: string }) {
  const filteredUsers = useMemo(() => {
    return users.filter(user =>

        <div key={user.id} onClick={() => handleDelete(user.id)}>
```

### 2.2 自定义Hooks

#### 【强制】自定义Hook命名

- Hook函数名以 `use` 开头
- Hook必须遵循Hook规则（只能在组件顶层调用）

#### ✅ 正确示例

```typescript
function useAuth() {
  const [user, setUser] = useState<User | null>(null);
  const [isLoading, setIsLoading] = useState<boolean>(true);

  useEffect(() => {
    const init = async () => {
      const token = storage.getToken();
      if (token) {
        try {
          const response = await api.getUserInfo();
          setUser(response.data);
        } catch (error) {
          storage.removeToken();
        }
      }
      setIsLoading(false);
    };

    init();
  }, []);

  const login = async (username: string, password: string) => {
    const response = await api.login({ username, password });
    setUser(response.data.user);
    storage.setToken(response.data.token);
  };

  const logout = () => {
    setUser(null);
    storage.removeToken();
  };

  return { user, isLoading, login, logout };
}

function useRequest<T>(url: string, options?: RequestOptions) {
  const [data, setData] = useState<T | null>(null);
  const [isLoading, setIsLoading] = useState<boolean>(false);
  const [error, setError] = useState<Error | null>(null);

  const fetchData = async () => {
    setIsLoading(true);
    setError(null);
    try {
      const response = await axios.get<T>(url, options);
      setData(response.data);
    } catch (err) {
      setError(err as Error);
    } finally {
      setIsLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
  }, [url, options]);

  return { data, isLoading, error, refetch: fetchData };
}

function useLocalStorage<T>(key: string, initialValue: T) {
  const [storedValue, setStoredValue] = useState<T>(() => {
    try {
      const item = localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch {
      return initialValue;
    }
  });

  const setValue = (value: T | ((val: T) => T)) => {
    try {
      const valueToStore = value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      localStorage.setItem(key, JSON.stringify(valueToStore));
    } catch (error) {
      console.error('Error saving to localStorage:', error);
    }
  };

  return [storedValue, setValue] as const;
}
```

## 3. Zustand状态管理规范

### 3.1 Store定义

#### 【强制】Store文件命名

- Store文件使用 `.store.ts` 后缀
- Store目录位于 `src/stores/`

#### 【强制】Store定义规范

```typescript
import { create } from 'zustand';

interface AuthStore {
  user: User | null;
  token: string | null;
  isLoggedIn: boolean;
  login: (user: User, token: string) => void;
  logout: () => void;
  setUser: (user: User | null) => void;
}

export const useAuthStore = create<AuthStore>((set, get) => ({
  user: null,
  token: localStorage.getItem('token') || null,
  isLoggedIn: () => !!get().token,
  
  login: (user, token) => {
    localStorage.setItem('token', token);
    set({ user, token, isLoggedIn: true });
  },
  
  logout: () => {
    localStorage.removeItem('token');
    set({ user: null, token: null, isLoggedIn: false });
  },
  
  setUser: (user) => {
    set({ user });
  },
}));
```

### 3.2 Store拆分

#### 【推荐】按业务模块拆分Store

```
src/stores/
├── auth.store.ts       # 认证状态
├── app.store.ts        # 应用状态（主题、布局等）
├── user.store.ts       # 用户相关状态
└── index.ts            # 统一导出
```

#### ✅ 正确示例

```typescript

  logout: () => void;

    localStorage.setItem('token', token);
    set({ user, token });
  },
  
  logout: () => {
    localStorage.removeItem('token');
    set({ user: null, token: null });
  },

// app.store.ts
interface AppStore {
  theme: 'light' | 'dark';
  sidebarCollapsed: boolean;
  setTheme: (theme: 'light' | 'dark') => void;
  toggleSidebar: () => void;
}

export const useAppStore = create<AppStore>((set) => ({
  theme: localStorage.getItem('theme') as 'light' | 'dark' || 'light',
  sidebarCollapsed: false,
  
  setTheme: (theme) => {
    localStorage.setItem('theme', theme);
    set({ theme });
  },
  
  toggleSidebar: () => set((state) => ({ sidebarCollapsed: !state.sidebarCollapsed })),
}));
```

### 3.3 Store使用

#### 【强制】Store使用规范

- 使用 `useStore` 获取状态
- 使用 `shallow` 进行选择器优化

```tsx
import { shallow } from 'zustand';

function UserProfile() {
  const { user, logout } = useAuthStore();

  if (!user) {
    return <div>Please login</div>;
  }

  return (
    <div>
      <h1>{user.username}</h1>
      <p>{user.email}</p>
      <button onClick={logout}>Logout</button>
    </div>
  );
}

function Sidebar() {
  const { sidebarCollapsed, toggleSidebar } = useAuthStore(
    (state) => ({
      sidebarCollapsed: state.sidebarCollapsed,
      toggleSidebar: state.toggleSidebar,
    }),
    shallow
  );

  return (
    <aside className={sidebarCollapsed ? 'collapsed' : ''}>
      <button onClick={toggleSidebar}>Toggle</button>
    </aside>
  );
}
```

### 3.4 异步操作

#### 【推荐】异步操作处理

```typescript
interface UserStore {
  users: User[];
  isLoading: boolean;
  error: string | null;
  fetchUsers: () => Promise<void>;
  createUser: (user: CreateUserRequest) => Promise<void>;
  updateUser: (id: number, user: UpdateUserRequest) => Promise<void>;
  deleteUser: (id: number) => Promise<void>;
}

export const useUserStore = create<UserStore>((set) => ({
  users: [],
  isLoading: false,
  error: null,
  
  fetchUsers: async () => {
    set({ isLoading: true, error: null });
    try {
      const response = await api.getUserList();
      set({ users: response.data });
    } catch (error) {
      set({ error: 'Failed to fetch users' });
    } finally {
      set({ isLoading: false });
    }
  },
  
  createUser: async (user) => {
    set({ isLoading: true, error: null });
    try {
      await api.createUser(user);
      await get().fetchUsers();
    } catch (error) {
      set({ error: 'Failed to create user' });
    } finally {
      set({ isLoading: false });
    }
  },
}));
```

## 4. 状态管理最佳实践

### 4.1 状态分层

#### 【强制】状态分层管理

| 状态类型 | 管理方式 | 适用场景 |
|---------|---------|---------|
| **组件状态** | useState | 单个组件内部状态 |
| **页面状态** | useState/useReducer | 复杂页面状态 |
| **全局状态** | Zustand | 用户登录、主题、布局 |
| **服务器状态** | useRequest/SWR/React Query | API数据缓存 |

### 4.2 避免状态冗余

#### 【推荐】计算状态

```tsx
function UserList({ users }: { users: User[] }) {
  const activeCount = useMemo(() => {
    return users.filter(user => user.status === 'active').length;
  }, [users]);

  return (
    <div>
      <div>Total: {users.length}</div>
      <div>Active: {activeCount}</div>
    </div>
  );
}
```

### 4.3 状态更新规范

#### 【强制】不可变更新

```tsx
function UserList() {
  const [users, setUsers] = useState<User[]>([]);

  const addUser = (user: User) => {
    setUsers(prev => [...prev, user]);
  };

  const updateUser = (id: number, updates: Partial<User>) => {
    setUsers(prev => prev.map(user => 
      user.id === id ? { ...user, ...updates } : user
    ));
  };

  const deleteUser = (id: number) => {
    setUsers(prev => prev.filter(user => user.id !== id));
  };

  return <div>{users.map(user => <div key={user.id}>{user.username}</div>)}</div>;
}
```
