## 2. 样式规范

### 2.1 CSS Modules

#### 【强制】CSS Modules使用

- 组件样式必须使用CSS Modules
- 样式文件使用 `.module.less` 后缀

```tsx
import styles from './UserList.module.less';

export default function UserList() {
  return (
    <div className={styles.container}>
      <h1 className={styles.title}>User List</h1>
      <div className={styles.userItem}>
        <span className={styles.userName}>张三</span>
      </div>
    </div>
  );
}
```

#### 【强制】CSS Modules命名

- 使用小驼峰命名
- 使用BEM风格

```less
.container {
  padding: 16px;
}

.title {
  font-size: 18px;
  font-weight: bold;
}

.userItem {
  display: flex;
  align-items: center;
  padding: 8px 0;
  
  &:hover {
    background-color: #f5f5f5;
  }
}

.userName {
  font-size: 14px;
}
```

## 3. 性能规范

### 3.1 代码分割

#### 【强制】路由级代码分割

```tsx
import { lazy, Suspense } from 'react';
import { BrowserRouter, Routes, Route } from 'react-router-dom';

const Login = lazy(() => import('./pages/login/Login'));
const Dashboard = lazy(() => import('./pages/dashboard/Dashboard'));
const UserList = lazy(() => import('./pages/user/list/UserList'));
const UserForm = lazy(() => import('./pages/user/form/UserForm'));

function App() {
  return (
    <BrowserRouter>
      <Suspense fallback={<div>Loading...</div>}>
        <Routes>
          <Route path="/login" element={<Login />} />
          <Route path="/dashboard" element={<Dashboard />} />
          <Route path="/user/list" element={<UserList />} />
          <Route path="/user/form" element={<UserForm />} />
        </Routes>
      </Suspense>
    </BrowserRouter>
  );
}
```

#### 【推荐】组件级代码分割

```tsx
import { lazy, Suspense } from 'react';

const ChartComponent = lazy(() => import('./ChartComponent'));

function Dashboard() {
  return (
    <div>
      <h1>Dashboard</h1>
      <Suspense fallback={<div>Loading chart...</div>}>
        <ChartComponent />
      </Suspense>
    </div>
  );
}
```

### 3.2 React.memo

#### 【强制】组件缓存

```tsx
interface UserCardProps {
  user: User;
  onEdit: (id: number) => void;
}

const UserCard = React.memo(({ user, onEdit }: UserCardProps) => {
  return (
    <div className="user-card">
      <h3>{user.username}</h3>
      <p>{user.email}</p>
      <button onClick={() => onEdit(user.id)}>Edit</button>
    </div>
  );
});

function UserList({ users, onEdit }: { users: User[]; onEdit: (id: number) => void }) {
  return (
    <div>
      {users.map(user => (
        <UserCard key={user.id} user={user} onEdit={onEdit} />
      ))}
    </div>
  );
}
```

### 3.3 useMemo/useCallback

#### 【强制】计算属性缓存

```tsx
function UserList({ users, filter }: { users: User[]; filter: string }) {
  const filteredUsers = useMemo(() => {
    return users.filter(user => 
      user.username.toLowerCase().includes(filter.toLowerCase())
    );
  }, [users, filter]);

  const handleDelete = useCallback((id: number) => {
    console.log('Delete user:', id);
  }, []);

  return (
    <div>
      {filteredUsers.map(user => (
        <div key={user.id} onClick={() => handleDelete(user.id)}>
          {user.username}
        </div>
      ))}
    </div>
  );
}
```

### 3.4 列表渲染优化

#### 【强制】列表渲染优化

```tsx
function UserList({ users }: { users: User[] }) {
  return (
    <div>
      {users.map(user => (
        <div key={user.id}>
          <span>{user.id}</span>
          <span>{user.username}</span>
          <span>{user.email}</span>
        </div>
      ))}
    </div>
  );
}

// 大数据量列表使用虚拟列表
import { List } from 'react-window';

function VirtualUserList({ users }: { users: User[] }) {
  const Row = ({ index, style }: { index: number; style: React.CSSProperties }) => {
    const user = users[index];
    return (
      <div style={style}>
        <span>{user.id}</span>
        <span>{user.username}</span>
      </div>
    );
  };

  return (
    <List
      height={400}
      itemCount={users.length}
      itemSize={50}
      width="100%"
    >
      {Row}
    </List>
  );
}
```

### 3.5 图片优化

#### 【推荐】图片懒加载

```tsx
import { LazyLoadComponent } from 'react-lazyload';

function ImageGallery({ images }: { images: string[] }) {
  return (
    <div>
      {images.map((src, index) => (
        <LazyLoadComponent key={index} height={200} offset={100}>
          <img src={src} alt={`Image ${index}`} />
        </LazyLoadComponent>
      ))}
    </div>
  );
}

// 使用loading占位
function Avatar({ src, alt }: { src: string; alt: string }) {
  return (
    <div className="avatar-wrapper">
      <img 
        src={src} 
        alt={alt} 
        loading="lazy"
        onLoad={(e) => {
          (e.target as HTMLImageElement).style.opacity = '1';
        }}
        style={{ opacity: 0 }}
      />
      <div className="avatar-loading" />
    </div>
  );
}
```

### 3.6 避免不必要的重渲染

#### 【强制】状态管理优化

```tsx
function Counter() {
  const [count, setCount] = useState(0);
  const [name, setName] = useState('');

  const doubledCount = useMemo(() => {
    return count * 2;
  }, [count]);

  return (
    <div>
      <div>Count: {count}</div>
      <div>Doubled: {doubledCount}</div>
      <input value={name} onChange={(e) => setName(e.target.value)} />
      <button onClick={() => setCount(prev => prev + 1)}>Increment</button>
    </div>
  );
}
```

## 4. 代码优化规范

### 4.1 避免内联函数

#### 【推荐】使用useCallback

```tsx
// ✅ 正确：使用useCallback
function UserList({ users, onEdit }: { users: User[]; onEdit: (id: number) => void }) {
  const handleEdit = useCallback((id: number) => {
    onEdit(id);
  }, [onEdit]);

  return (
    <div>
      {users.map(user => (
        <button key={user.id} onClick={() => handleEdit(user.id)}>
          Edit {user.username}
        </button>
      ))}
    </div>
  );
}

// ❌ 错误：内联函数会导致每次渲染都创建新函数
function BadUserList({ users, onEdit }: { users: User[]; onEdit: (id: number) => void }) {
  return (
    <div>
      {users.map(user => (
        <button key={user.id} onClick={() => onEdit(user.id)}>
          Edit {user.username}
        </button>
      ))}
    </div>
  );
}
```

### 4.2 避免内联样式

#### 【强制】使用CSS Modules

```tsx
// ✅ 正确：使用CSS Modules
import styles from './Button.module.less';

function Button({ children, type = 'primary' }: { children: React.ReactNode; type?: 'primary' | 'secondary' }) {
  return (
    <button className={`${styles.button} ${styles[type]}`}>
      {children}
    </button>
  );
}

// ❌ 错误：内联样式难以维护
function BadButton({ children }: { children: React.ReactNode }) {
  return (
    <button style={{ backgroundColor: '#1890ff', color: '#fff', padding: '8px 16px' }}>
      {children}
    </button>
  );
}
```
